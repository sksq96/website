I am learning about NFTs. As part of my learning process, I am trying to build an NFT marketplace. Why? What are NFTs? What is a marketplace? Let me share my journey with you on my learning process.

The first time I heard about NFTs was on social media (Twitter, Youtube, Tiktok, Instagram). I was curious to learn more about them and I started doing some research. I found about many articles explaining the concept but it was not until I started building something myself that I felt comfortable about my understanding of the technology.

I can define NFTs as follows:

> NFTs are a technology that allow us to have verifiable ownership of a digital asset. The digital asset can be anything from art, music, photos, etc. Also, NFTs are blockchain-based tokens that can be traded, bought and sold.

Let's go over the architecture of the marketplace.

![Architecture](https://i.imgur.com/1fML0T7.png)

The marketplace is a decentralized app. It is an application where you can buy/sell NFTs. The web app talks with the blockchain to interact with the marketplace and the NFT.

In blockchain, we have smart contracts. A smart contract is an executable program that lives on the blockchain. These contracts execute when a transaction is made. In our marketplace dApp we have two contracts:

- Marketplace: this contract allows people to interact with the marketplace (i.e. buy/sell NFTs)
- NFT: this contract handles the NFT itself (i.e. minting the NFT)

The web application is built using Next.js and the smart contracts are written with Solidity.

The marketplace contract is shown below

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "@openzeppelin/contracts/token/ERC721/IERC721.sol";

contract NFTMarket {

    uint256 private _memos;
    uint256 private _items;
    uint256 public listingPrice = 2e15; // 0.002 eth

    // A struct is a custom data type that we can define
    // This struct will be used to represent a created item.
    // It will be used to keep track of the item's name,
    // description, price, creator, and current owner.
    struct Item {
        string name;
        string description;
        uint256 price;
        address nftContract;
        uint256 tokenId;
        uint256 itemId;
        address payable seller;
        bool sold;
    }

    event ItemCreated(
        string name,
        string description,
        uint256 price,
        address nftContract,
        uint256 tokenId,
        uint256 itemId,
        address seller
    );

    // map the itemId to the ite
    mapping(uint256 => Item) public items;

    function createItem(uint256 price, address nftContract, uint256 tokenId) public payable {

        require(price > 0, "Price must be at least 1 wei");
        require(msg.value == listingPrice, "Must pay listing price");

        _items++;
        uint256 itemId = _items;

        items[itemId] = Item({
            name: "NFT",
            description: "NFT description",
            price: price,
            nftContract: nftContract,
            tokenId: tokenId,
            itemId: itemId,
            seller: payable(msg.sender),
            sold: false
        });

        IERC721(nftContract).transferFrom(msg.sender, address(this), tokenId);

        emit ItemCreated("NFT", "NFT description", price, nftContract, tokenId, itemId, msg.sender);
    }

    function purchaseItem(uint256 itemId) public payable {
        uint256 price = items[itemId].price;
        uint256 tokenId = items[itemId].tokenId;
        address nftContract = items[itemId].nftContract;
        address payable seller = items[itemId].seller;

        require(msg.value == price, "Please submit the asking price in order to complete the purchase");

        items[itemId].seller = payable(msg.sender);
        items[itemId].sold = true;

        IERC721(nftContract).transferFrom(address(this), msg.sender, tokenId);

        payable(address(this)).transfer(listingPrice);
        payable(seller).transfer(msg.value);
    }

    function fetchItemsForSale() public view returns (Item[] memory) {
        uint256 itemCount = _items;
        uint256 unsoldItemCount = 0;
        uint256 currentIndex = 0;

        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].sold == false) {
                unsoldItemCount += 1;
            }
        }

        Item[] memory itemsForSale = new Item[](unsoldItemCount);
        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].sold == false) {
                uint256 currentId = i + 1;
                Item storage currentItem = items[currentId];
                itemsForSale[currentIndex] = currentItem;
                currentIndex += 1;
            }
        }
        return itemsForSale;
    }

    function fetchMyNFTs() public view returns (Item[] memory) {
        uint256 itemCount = _items;
        uint256 myItemCount = 0;
        uint256 currentIndex = 0;

        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].seller == msg.sender) {
                myItemCount += 1;
            }
        }

        Item[] memory myItems = new Item[](myItemCount);
        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].seller == msg.sender) {
                uint256 currentId = i + 1;
                Item storage currentItem = items[currentId];
                myItems[currentIndex] = currentItem;
                currentIndex += 1;
            }
        }
        return myItems;
    }

    function fetchItemsCreated() public view returns (Item[] memory) {
        uint256 itemCount = _items;
        uint256 myItemCount = 0;
        uint256 currentIndex = 0;

        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].seller == msg.sender) {
                myItemCount += 1;
            }
        }

        Item[] memory myItems = new Item[](myItemCount);
        for (uint256 i = 0; i < itemCount; i++) {
            if (items[i + 1].seller == msg.sender) {
                uint256 currentId = i + 1;
                Item storage currentItem = items[currentId];
                myItems[currentIndex] = currentItem;
                currentIndex += 1;
            }
        }
        return myItems;
    }
}
```

It contains 3 functions:

- createItem: creates an item that can be sold on the marketplace. It requires that the NFT is transferred to the contract first.
- purchaseItem: allows someone to purchase an item that is listed on the marketplace. It transfers the NFT from the contract to the buyer.
- fetchItemsForSale: returns all the items that are currently for sale.

The NFT contract is show below

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";

contract NFT is ERC721 {

    uint256 private _tokenIds;

    constructor() ERC721("NFT", "NFT") {}

    function createNFT() public returns (uint256) {
        _tokenIds++;
        uint256 newItemId = _tokenIds;
        _mint(msg.sender, newItemId);
        return newItemId;
    }
}
```

It contains the createNFT function which allows anyone to mint an NFT.

## Challenges

I found it very challenging to build the marketplace dApp. I had to learn many new concepts:

- Solidity, the programming language for writing smart contracts.
- ERC721, the standard for NFTs.
- Ethers.js, a library for interacting with the blockchain.
- IPFS, a protocol for storing and sharing files in a distributed file system.
- Metamask, a browser extension that allows you to interact with the blockchain.

I was able to overcome these challenges by reading the documentation and asking questions on Discord. I also had to debug many issues. I learned a lot by doing.
